// ===== §164 (23/11/2026): sub-aba "Ranking" — Linhas, Fibra e Soluções digitais, separados por operação =====
// Conta os pedidos pela data de cadastro (SP), sem Venda perdida, Devolvido, Proposta e Aguardando interação.
// Desempate pela receita do próprio ranking (aparelho nunca entra) e depois pelo nome. Apex e Campinas nunca se misturam.
var RANKING_CTX = { eu: null, equipe: [] };
try{ RANKING_CTX = __RANKING__ || RANKING_CTX; }catch(e){ RANKING_CTX = { eu: null, equipe: [] }; }
const RANKING_FORA = new Set(['ANDERSON TEIXEIRA MIZUTA']); // pedido do Rafael: o dono fica fora só do ranking
const RANKING_ETAPAS_FORA = new Set(['VENDA PERDIDA', 'DEVOLVIDO', 'PROPOSTA', 'AGUARDANDO INTERACAO']);
const RANKING_DIGITAIS = new Set(['SVA MOVEL', 'SVA FIXA', 'INCREMENTO DE MODULO', 'NET - TV']);
const RANKINGS = [
  { k: 'linhas', titulo: 'Linhas', crit: 'Linhas de voz, sem soluções digitais. Empate: maior receita das linhas.', un: ['linha', 'linhas'] },
  { k: 'fibra', titulo: 'Fibra', crit: 'Fibra e modem. Empate: maior receita da fibra.', un: ['fibra', 'fibras'] },
  { k: 'digitais', titulo: 'Soluções digitais', crit: 'Claro Monitor, Wi-Fi Mesh, Passaporte e Claro TV. Empate: maior receita.', un: ['solução', 'soluções'] },
];
function rkNorm(s){ return String(s == null ? '' : s).normalize('NFD').replace(/[\u0300-\u036f]/g, '').toUpperCase().trim(); }
function rankingCategoria(grupo){
  const g = rkNorm(grupo);
  if(g.indexOf('VOZ') === 0) return 'linhas';
  if(g.indexOf('BANDA LARGA') === 0) return 'fibra';
  return RANKING_DIGITAIS.has(g) ? 'digitais' : null;
}
function rankingConta(etapa){ return !RANKING_ETAPAS_FORA.has(rkNorm(etapa).replace(/\s*\(NEOCRM\)$/, '')); }
function rkSomaDias(dia, n){ const [y, m, d] = dia.split('-').map(Number); return new Date(Date.UTC(y, m - 1, d + n)).toISOString().slice(0, 10); }
function rankingPeriodo(tipo, hoje, mes){
  if(tipo === 'hoje') return { inicio: hoje, fim: hoje };
  if(tipo === 'semana'){
    const [y, m, d] = hoje.split('-').map(Number);
    const dow = new Date(Date.UTC(y, m - 1, d)).getUTCDay(); // 0 = domingo
    return { inicio: rkSomaDias(hoje, -((dow + 6) % 7)), fim: hoje };
  }
  const ym = mes || hoje.slice(0, 7);
  if(ym === hoje.slice(0, 7)) return { inicio: ym + '-01', fim: hoje };
  const [y, m] = ym.split('-').map(Number);
  return { inicio: ym + '-01', fim: new Date(Date.UTC(y, m, 0)).toISOString().slice(0, 10) };
}
function rankingCalcular(dados, opts){
  const por = new Map();
  const pega = nome => {
    let p = por.get(nome);
    if(!p){ p = { nome, linhas: { q: 0, r: 0, modem: 0 }, fibra: { q: 0, r: 0, modem: 0 }, digitais: { q: 0, r: 0, modem: 0 } }; por.set(nome, p); }
    return p;
  };
  (dados || []).forEach(r => {
    const nome = String((r && r.usuario) || '').trim().toUpperCase();
    if(!nome || RANKING_FORA.has(nome)) return;
    if(projOperacaoDaLinha(r, opts.operacao) !== opts.operacao) return;
    const cat = rankingCategoria(r.grupo);
    if(!cat || !rankingConta(r.etapa)) return;
    const dia = diaSP(r.cadastro);
    if(!dia || dia < opts.inicio || dia > opts.fim) return;
    const q = Number(r.quantidade) || 0, v = Number(r.valor) || 0;
    const p = pega(nome);
    p[cat].q += q; p[cat].r += v;
    if(cat === 'fibra' && tipoVendaDe(r) === 'BANDA LARGA - Modem') p.fibra.modem += q;
  });
  (opts.equipe || []).forEach(e => {
    const nome = String((e && e.nome) || '').trim().toUpperCase();
    if(nome && !RANKING_FORA.has(nome) && e.operacao === opts.operacao) pega(nome);
  });
  const out = {};
  RANKINGS.forEach(R => {
    out[R.k] = Array.from(por.values()).map(p => ({ nome: p.nome, q: p[R.k].q, r: p[R.k].r, modem: p[R.k].modem }))
      .sort((a, b) => b.q - a.q || b.r - a.r || a.nome.localeCompare(b.nome, 'pt-BR'));
  });
  return out;
}
function rankingPosicoes(lista){ const m = new Map(); (lista || []).forEach((x, i) => { if(x.q > 0) m.set(x.nome, i + 1); }); return m; }
function rankingSeta(posAgora, posAntes){
  if(!posAgora) return '';
  if(!posAntes) return 'novo';
  return posAntes > posAgora ? 'sobe' : posAntes < posAgora ? 'cai' : 'igual';
}
function rankingSuaPosicao(lista, nome){
  const i = (lista || []).findIndex(x => x.nome === nome);
  if(i < 0) return null;
  const me = lista[i];
  if(me.q === 0) return { pos: null, q: 0 };
